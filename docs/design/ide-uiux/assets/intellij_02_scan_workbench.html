package com.acme.payment.validator;
 
import org.springframework.stereotype.Service;
import org.springframework.validation.annotation.Validated;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import java.math.BigDecimal;
import java.util.Objects;
 
@Slf4j
@Service
@Validated
public class OrderPaymentValidator {
 
private final RiskAnalysisClient riskClient;
private final CurrencyWhitelistProvider currencyProvider;
 
@Override
public ValidationResult validatePayment(OrderRequest request) {
// Pre-check payload integrity and limits
Objects.requireNonNull(request, "Payment request payload is null");
BigDecimal amount = BigDecimal.valueOf(request.getRawAmount());
 
if (amount.compareTo(BigDecimal.ZERO) <= 0) {
throw new InvalidPaymentAmountException("Negative amount: " + amount);
}
 
if (!currencyProvider.isSupported(request.getCurrency())) {
return ValidationResult.rejected("Unsupported currency");
}
 
return riskClient.evaluateTransaction(request.getOrderId(), amount);
}
}
